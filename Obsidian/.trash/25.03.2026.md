	ВидДокумента = Справочники.ВидыДокументов.НайтиПоНаименованию("Заявка на расходование денежных средств");

	НашлиЗРДС = Ложь;
	НашлиПодходящуюЗРДС = Ложь;

	Для Каждого Стр Из Задача.Предметы Цикл
	
		Если Стр.Предмет.ВидДокумента = ВидДокумента Тогда
			НашлиЗРДС = Истина;

			СтатусКА2 = "";
			ЗакрытаКА2 = "";

			Для Каждого СтрДоп Из Стр.Предмет.ДополнительныеРеквизиты Цикл
				Если СтрДоп.Свойство.Наименование = "Статус из КА2" Тогда
					СтатусКА2 = СтрДоп.Значение;
				ИначеЕсли СтрДоп.Свойство.Наименование = "Закрыта из КА2" Тогда
					ЗакрытаКА2 = СтрДоп.Значение;
				КонецЕсли;
			КонецЦикла;

			Если ВРег(СокрЛП(СтатусКА2)) = ВРег("К оплате") И ВРег(СокрЛП(ЗакрытаКА2)) = ВРег("Да") Тогда
				НашлиПодходящуюЗРДС = Истина;
				Прервать; 
			КонецЕсли;

		КонецЕсли;

	КонецЦикла;

	Если НашлиПодходящуюЗРДС Тогда
		Результат = Ложь;
	Иначе
		Результат = Истина;

		Если НашлиЗРДС Тогда
			Сообщить("Документ найден, но статус/закрыт не подходят (нужен статус: 'К оплате' и Закрыта: 'Да').");
		Иначе
			Сообщить("Для продолжения, необходимо добавить заявку на расходование денежных средств");
		КонецЕсли;

	КонецЕсли;
